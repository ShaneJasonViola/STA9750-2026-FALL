# Acquire Netflix What We Watched Reports
load_netflix_data <- function() {

  # Create the folder where Netflix data will be saved
  data_dir <- fs::path("data", "mp01")

  if (!fs::dir_exists(data_dir)) {
    fs::dir_create(
      data_dir,
      showWarnings = FALSE,
      recurse = TRUE
    )
  }

  # Netflix reporting webpages
  NETFLIX_URLS <- list(
    `2023H1` = "https://about.netflix.com/news/what-we-watched-a-netflix-engagement-report",
    `2023H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2023",
    `2024H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2024",
    `2024H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2024",
    `2025H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2025",
    `2025H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2025",
    `2026H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2026"
  )

  # Specify the local Excel file paths
  local_paths <- fs::path(
    data_dir,
    paste0(names(NETFLIX_URLS), ".xlsx")
  )

  # Download only the files that are missing
  for (p in names(NETFLIX_URLS)) {

    local_file <- fs::path(
      data_dir,
      paste0(p, ".xlsx")
    )

    if (!fs::file_exists(local_file)) {

      webpage <- rvest::read_html(NETFLIX_URLS[[p]])

      excel_url <- webpage |>
        rvest::html_element(
          xpath = "//a[contains(@href, '.xlsx')]"
        ) |>
        rvest::html_attr("href")

      if (length(excel_url) != 1 || is.na(excel_url)) {
        stop("Excel file not found for ", p)
      }

      excel_url <- xml2::url_absolute(
        excel_url,
        NETFLIX_URLS[[p]]
      )

      download.file(
        excel_url,
        local_file,
        quiet = TRUE,
        mode = "wb"
      )
    }
  }

  # Read and combine the seven reports
  tibble(
    period = names(NETFLIX_URLS),
    path = local_paths
  ) |>
    mutate(
      data = map(
        path,
        \(x) read_excel(
          x,
          skip = 3,
          na = "*"
        )
      ),
      data = map(
        data,
        \(x) x |>
          mutate(
            `Release Date` = as.Date(`Release Date`)
          )
      )
    ) |>
    unnest(data) |>
    select(-path) |>

    # Create the reporting period dates
    mutate(
      year = str_sub(period, 1, 4),
      half = str_sub(period, start = 6),

      period_start = make_date(
        year,
        if_else(half == 1, 1, 7),
        1
      ),

      period_end = make_date(
        year,
        if_else(half == 1, 6, 12),
        if_else(half == 1, 30, 31)
      )
    ) |>

    # Exclude missing viewing hours
    filter(!is.na(`Hours Viewed`)) |>

    # Convert runtime to minutes
    mutate(
      live_event = !is.na(Runtime) &
        str_detect(Runtime, "\\d+:\\d+\\*"),

      clean_time = str_remove(
        Runtime,
        fixed("*")
      ),

      Runtime = 60 * hour(hm(clean_time)) +
        minute(hm(clean_time))
    ) |>
    select(-clean_time) |>

    # Clean runtime and title values
    mutate(
      Runtime = coalesce(
        Runtime,
        mean(Runtime, na.rm = TRUE)
      ),

      .by = Title,

      Title = stringi::stri_trans_general(
        Title,
        "Latin-ASCII"
      ),

      Title = str_remove(Title, " // .*"),
      Title = str_remove(Title, "[^[:ascii:]].*"),
      Title = str_trim(Title)
    ) |>
    filter(Title != "") |>

    # Keep the columns needed for the project
    select(
      year,
      period,
      period_start,
      period_end,
      Title,
      -half,
      Runtime,
      Views,
      `Hours Viewed`,
      `Release Date`,
      `Available Globally?`,
      live_event
    )
}

make_season_data <- function(
    prefix,
    seasons,
    before,
    during
) {

  NETFLIX_DATA |>
    filter(
      period %in% c(before, during),

      str_detect(
        title,
        regex(
          paste0(
            "^",
            prefix,
            "(?:\\s*:?\\s*(?:Season\\s*)?[1-",
            seasons,
            "])?$"
          ),
          ignore_case = TRUE
        )
      )
    ) |>
    mutate(
      season_number = as.integer(
        str_extract(
          title,
          paste0("[1-", seasons, "]$")
        )
      ),

      season_number = coalesce(
        season_number,
        1L
      )
    ) |>
    group_by(season_number, period) |>
    summarise(
      viewing_hours = sum(
        hours_viewed,
        na.rm = TRUE
      ),
      .groups = "drop"
    ) |>
    mutate(
      season_label = factor(
        paste0("Season ", season_number),
        levels = paste0(
          "Season ",
          seasons:1
        )
      ),

      reporting_window = factor(
        period,
        levels = c(before, during),
        labels = c(
          "Previous period",
          "New-season period"
        )
      ),

      viewing_millions = viewing_hours / 1000000
    )
}

# 1. calculate the Gini coefficient


calculate_gini <- function(x) {

  x <- x[is.finite(x) & x >= 0]

  if (length(x) == 0 || sum(x) == 0) {
    return(NA_real_)
  }

  x <- sort(x)

  n <- length(x)

  (2 * sum(seq_along(x) * x) /
     (n * sum(x))) - (n + 1) / n
}
